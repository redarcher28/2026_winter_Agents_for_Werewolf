## AI Agent可用接口文档

### 1. WebSocket连接接口

```python
# 连接端点
ws://judge-server/game/{game_id}/agent/{agent_id}

# 连接参数
headers = {
    "Authorization": "Bearer {agent_token}",
    "X-Agent-Role": "{role_name}",
    "X-Game-Version": "1.0"
}

# 心跳机制
# 客户端每30秒发送ping，服务器回应pong
{
    "type": "ping",
    "timestamp": "2024-01-20T10:00:00Z"
}
```

### 2. 游戏状态查询接口

```python
# 请求公共游戏信息
{
    "request_id": "req_001",
    "action": "query_public_state",
    "params": {}
}

# 响应示例
{
    "response_to": "req_001",
    "data": {
        "game_id": "game_123",
        "phase": "daytime_discussion",
        "day_number": 3,
        "alive_players": [
            {"id": "p1", "name": "玩家1", "is_ai": true},
            {"id": "p2", "name": "玩家2", "is_ai": false}
        ],
        "dead_players": [
            {"id": "p3", "name": "玩家3", "death_reason": "狼人击杀"}
        ],
        "time_remaining": 120  # 秒
    }
}

# 请求角色特定信息（权限过滤后）
{
    "request_id": "req_002",
    "action": "query_role_info",
    "params": {
        "info_type": "werewolf_partners"  # 或 "seer_check_history", "witch_potions"
    }
}
```

## 接口使用示例代码

```python
class SimpleWerewolfAgent:
    def __init__(self, agent_id, game_id, server_url):
        self.agent_id = agent_id
        self.game_id = game_id
        self.ws_client = WebSocketClient(server_url)
        self.memory = AgentMemory()
        self.llm = LLMClient()
        
    async def play_game(self):
        # 1. 连接到游戏
        await self.ws_client.connect(
            f"{self.game_id}/agent/{self.agent_id}"
        )
        
        # 2. 监听游戏事件
        async for event in self.ws_client.listen():
            self.memory.add_event(event)
            
            if event.type == "phase_change":
                await self.handle_phase_change(event)
            elif event.type == "player_speech":
                await self.analyze_speech(event)
            elif event.type == "vote_result":
                await self.update_strategy(event)
                
    async def handle_phase_change(self, event):
        """处理阶段变更"""
        new_phase = event.data["new_phase"]
        
        if new_phase == "werewolf_night":
            # 狼人夜晚行动
            await self.take_werewolf_night_action()
        elif new_phase == "daytime_discussion":
            # 白天发言
            await self.participate_discussion()
        elif new_phase == "daytime_voting":
            # 投票
            await self.cast_vote()
            
    async def take_werewolf_night_action(self):
        """狼人夜晚杀人决策"""
        # 查询同伴信息
        partners = await self.ws_client.query("werewolf_partners")
        
        # 查询可杀目标
        game_state = await self.ws_client.query("public_state")
        alive_players = game_state["alive_players"]
        
        # LLM决策选择目标
        decision_context = {
            "partners": partners,
            "alive_players": alive_players,
            "game_history": self.memory.get_summary()
        }
        
        target = await self.llm.decide_kill_target(decision_context)
        
        # 提交行动
        await self.ws_client.submit_action({
            "action_type": "kill",
            "target": target
        })
```
